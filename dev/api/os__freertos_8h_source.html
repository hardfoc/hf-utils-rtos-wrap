114static inline OS_Uint os_thread_create(OS_Thread *t, const char *name,
115 void (*entry)(OS_Ulong),
116 OS_Ulong input, uint8_t *stack,
117 OS_Ulong stack_size, OS_Uint priority,
118 OS_Uint preempt, OS_Ulong slice,
119 OS_Uint auto_start)
120{
121 /* xTaskCreate allocates the stack. stack/preempt/slice stay in the
122 * ThreadX-shaped signature; this backend does not install them. */
123 (void)stack;
124 (void)preempt;
125 (void)slice;
126 os_thread_start_t *params =
127 (os_thread_start_t *)pvPortMalloc(sizeof(os_thread_start_t));
128 if (!params) return 1;
129 params->entry = entry;
130 params->arg = input;
131 BaseType_t res = xTaskCreate(os_thread_start_trampoline, name,
132 stack_size / sizeof(StackType_t), params,
133 priority, t);
134 if (res != pdPASS) {
135 vPortFree(params);
136 return 1;
137 }
138 if (!auto_start)
139 vTaskSuspend(*t);
140 return OS_SUCCESS;
141}
