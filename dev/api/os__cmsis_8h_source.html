83static inline OS_Uint os_thread_spawn(OS_Thread *t, const char *name,
84 void (*entry)(void *), void *arg,
85 void *stack, OS_Ulong stack_size,
86 OS_Uint priority, OS_Uint auto_start)
87{
88 if (t == 0 || entry == 0) {
89 return (OS_Uint)1;
90 }
91 osThreadAttr_t attr = OS_ZERO_INIT;
92 attr.name = name;
93 attr.stack_size = (uint32_t)stack_size;
94 attr.stack_mem = stack;
95 if (priority != 0u) {
96 attr.priority = (osPriority_t)priority;
97 }
98 osThreadId_t id = osThreadNew(entry, arg, &attr);
99 if (id == 0) {
100 return (OS_Uint)1;
101 }
102 *t = id;
103 if (!auto_start) {
104 (void)osThreadSuspend(id);
105 }
106 return OS_SUCCESS;
107}
