106 void Publish(const T& value) noexcept override
107 {

109 seq_.store(s0 + 1U, std::memory_order_release); // mark odd (writing)

111 std::memcpy(&payload_, &value, sizeof(T));
112 std::atomic_thread_fence(std::memory_order_release);
113 seq_.store(s0 + 2U, std::memory_order_release); // back to even
114 SignalChange_();

123 if ((s1 & 1U) != 0U) {
