49/* ── Clock ─────────────────────────────────────────────────────────────
50 * Without a scheduler there is no tick, so by default time stands still:
51 * os_time_get() returns 0 and delays return at once.
52 *
53 * Define HF_RTOS_NONE_EXTERNAL_CLOCK to take time from the integrator
54 * instead. The integrator then provides the two functions below (a host
55 * test build backs them with a fake clock; a bare-metal image could back
56 * them with a hardware timer). One tick is one millisecond, matching
57 * os_convert_msec_to_delay_ticks() for this backend. */
58#if defined(HF_RTOS_NONE_EXTERNAL_CLOCK)
60uint64_t hf_os_none_clock_us(void);
62void hf_os_none_delay_us(uint64_t us);
63static inline OS_Ulong os_time_get(void) { return (OS_Ulong)(hf_os_none_clock_us() / 1000U); }
64#else
65static inline OS_Ulong os_time_get(void) { return 0; }
66#endif
