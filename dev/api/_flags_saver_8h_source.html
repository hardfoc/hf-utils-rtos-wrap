160 bool ClearAll(uint32_t now_ms = 0) noexcept override

204 bool WaitForChange(uint32_t timeout_ms) noexcept override
205 {
206 if (!EnsureEventGroup_()) return false;
207 OS_Ulong actual = 0;
208 const OS_Ulong wait = (timeout_ms == UINT32_MAX)
209 ? static_cast<OS_Ulong>(OS_WAIT_FOREVER)
210 : static_cast<OS_Ulong>(os_convert_msec_to_delay_ticks(timeout_ms));
211 // Consume the signal: the next WaitForChange() waits for a new change.
212 const OS_Uint rc = os_event_group_get(&event_group_, kEventBit_,
213 static_cast<OS_Uint>(OS_OR) |
214 static_cast<OS_Uint>(OS_EVENT_CLEAR),
215 &actual, wait);
216 return (rc == OS_SUCCESS) && ((actual & kEventBit_) != 0U);
217 }

219 bool ClearWaitEvent() noexcept override
220 {
221 if (!EnsureEventGroup_()) return false;
222 return os_event_group_clear(&event_group_, kEventBit_) == OS_SUCCESS;
223 }

225private:
226 bool WriteSlot_(FlagId id, bool set, uint32_t now_ms) noexcept
227 {
228 const std::size_t idx = static_cast<std::size_t>(id);
229 if (idx >= kCount) return false;
230
231 const std::size_t word = idx / 64U;
232 const uint64_t mask = uint64_t{1} << (idx % 64U);
233 auto& w = words_[word];
234
235 const uint64_t prev = set
236 ? w.fetch_or(mask, std::memory_order_acq_rel)
237 : w.fetch_and(~mask, std::memory_order_acq_rel);
238
239 const bool was_set = (prev & mask) != 0U;
240 if (was_set == set) return false; // no transition
241
242 if (now_ms != 0U) last_change_ms_.store(now_ms, std::memory_order_release);
243 SignalChange_();
244 return true;
245 }

247 void SignalChange_() noexcept
248 {
249 seq_.fetch_add(1, std::memory_order_acq_rel);
250 if (EnsureEventGroup_()) {
251 (void)os_event_group_set(&event_group_, kEventBit_);
252 }
253 }
254
255 bool EnsureEventGroup_() noexcept
256 {
257 if (event_group_created_) return true;
258 if (os_event_group_create(&event_group_, "FlagsSaver") == OS_SUCCESS) {
259 event_group_created_ = true;
260 }
261 return event_group_created_;
262 }
263
264 static constexpr OS_Ulong kEventBit_ = 0x1U;
265
266 std::atomic<uint64_t> words_[kWordCount]{};
267 std::atomic<uint32_t> seq_{0};
268 std::atomic<uint32_t> last_change_ms_{0};
269 OS_EventGroup event_group_{};
270 bool event_group_created_{false};
271};
