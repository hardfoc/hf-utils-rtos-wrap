200template <typename MutexT>
201class RtosUniqueLock {
202public:
208 explicit RtosUniqueLock(MutexT& mutex, uint32_t timeout_ms = 0) noexcept
209 : mutex_(&mutex), locked_(false) {
210 if (timeout_ms > 0) {
211 locked_ = mutex_->try_lock_for(timeout_ms);
212 } else {
213 locked_ = mutex_->lock();
214 }
215 }
216
217 ~RtosUniqueLock() noexcept {
218 if (locked_ && mutex_) {
219 mutex_->unlock();
220 }
221 }

229 }

232 if (this != &other) {

234 mutex_->unlock();
